    {

            System.out.println("1-Doctor     2-Nurse     3-Back to previous menu");
            int choice = MainSystem.GetChoice("Choose: ", 1, 3);
            if (choice == 3) return;

            String name = "";
            String speciality = "";
            double salary = 5000.0;
            String username = "";
            String password = "";

            int step = 1;
            while (step != 0) {
                switch (step) {
                    case 1 -> {
                        System.out.println("Type \"back\" to go to the previous step.");
                        name = MainSystem.GetInput("Enter your name: ");
                        if (name.equalsIgnoreCase("back")) {
                            step = -1;
                        } else {
                            System.out.println("Your name is " + name + ".");
                            step = (choice == 1) ? 2 : 3;
                        }
                    }
                    case 2 -> {
                        speciality = MainSystem.GetInput("Enter your speciality: ");
                        if (speciality.equalsIgnoreCase("back")) {
                            step = 1;
                        } else {
                            System.out.println("Your Speciality is " + speciality + ".");
                            step = 3;
                        }
                    }
                    case 3 -> {
                        username = GetNewUsername();
                        if (username.equalsIgnoreCase("back")) {
                            step = (choice == 1) ? 2 : 1;
                        } else {
                            step = 4;
                        }
                    }
                    case 4 -> {
                        password = GetNewPassword();
                        if (password.equalsIgnoreCase("back")) {
                            step = 3;
                        } else {
                            step = 0;
                        }
                    }
                }
                if (step == -1) break;
            }

            if (step == 0)
            {
                if (choice == 1)
                {
                    e.add(new Doctor(name, salary, new Account(username, password), speciality));
                } else
                {
                    e.add(new Nurse(name, salary, new Account(username, password)));
                }
                return;
            }
        }
    }
}

// This file is responsible for patient and room logic
import java.util.ArrayList;

public class Patient
{
    public static int id_counter = 0;
    public static final double CONSULTATION_FEE = 200.0;
    public static final double ROOM_FEE = 500.0;

    String name;
    String medical_history;
    int id;
    Doctor assigned_doctor;
    Room reserved_room;
    double hospital_fees;

    Patient(String n, String m, Doctor d, Room r)
    {
        name = n;
        medical_history = m;
        assigned_doctor = d;
        id_counter++;
        id = id_counter;
        reserved_room = r;
        hospital_fees = CONSULTATION_FEE;

        if (r != null) {
            r.isFree = false;
            hospital_fees += ROOM_FEE;
            Room.sortRooms(DataBase.rooms);
        }
    }


    Patient(String n, String m, Doctor d)
    {
        this(n, m, d, null);
    }


    public void addFee(double amount) {
        if (amount > 0) {
            hospital_fees += amount;
        }
    }


    public boolean payFees(double amount) {
        if (amount <= 0) {
            System.out.println("Invalid payment amount.");
            return false;
        }
        if (amount > hospital_fees) {
            System.out.printf("Amount exceeds current fees ($%.2f).%n", hospital_fees);
            return false;
        }
        hospital_fees -= amount;
        System.out.printf("Paid $%.2f. Remaining balance: $%.2f%n", amount, hospital_fees);
        return true;
    }


    public static int searchPatientIndexById(ArrayList<Patient> patients, int targetId) {
        for (int i = 0; i < patients.size(); i++) {
            if (patients.get(i).id == targetId) {
                return i;
            }
        }
        return -1;
    }
}

class Room {
    public static final int MAX_ROOMS = 30;

    public int roomNumber;
    public boolean isFree;

    public Room(int roomNumber) {
        this.roomNumber = roomNumber;
        this.isFree = true;
    }

    public static Room[] initRooms() {
        Room[] rooms = new Room[MAX_ROOMS];
        for (int i = 0; i < MAX_ROOMS; i++) {
            rooms[i] = new Room(i + 1);
        }
        return rooms;
    }

    public static boolean assignFirstFreeRoom(Patient p) {
        if (!DataBase.rooms[0].isFree) {
            System.out.println("No free rooms available.");
            return false;
        }

        if (p.reserved_room != null) {
            releaseRoom(p);
        }

        Room chosen = DataBase.rooms[0];
        chosen.isFree = false;
        p.reserved_room = chosen;
        p.addFee(Patient.ROOM_FEE);

        sortRooms(DataBase.rooms);
        return true;
    }

    public static void releaseRoom(Patient p) {
        if (p.reserved_room != null) {
            p.reserved_room.isFree = true;
            p.reserved_room = null;
            sortRooms(DataBase.rooms);
        }
    }

    // Rooms are sorted by selection sort
    public static void sortRooms(Room[] rooms) {
        for (int i = 0; i < rooms.length - 1; i++) {
            int bestIdx = i;

            for (int j = i + 1; j < rooms.length; j++) {
                if ((rooms[j].isFree && !rooms[bestIdx].isFree) ||
                    (rooms[j].isFree == rooms[bestIdx].isFree && rooms[j].roomNumber < rooms[bestIdx].roomNumber)) {
                    bestIdx = j;
                }
            }

            Room temp = rooms[i];
            rooms[i] = rooms[bestIdx];
            rooms[bestIdx] = temp;
        }
    }
}

import java.util.ArrayList;

// the general class that all other employees will inherit from
public abstract class Employee
{
    protected static int id_generator = 0;
    protected String name;
    protected String jobtitle;
    protected int id;
    protected double salary;
    protected Account account;


    Employee(String n, double s, Account a)
    {
        name = n;
        id_generator++;
        id = this.id_generator;
        salary = s;
        account = a;
    }

    public abstract void GetEmployeeInfo();
}

    public static int GetChoice(String prompt)
    {
        System.out.print(prompt);
        String userinput;
        userinput = in.nextLine();

        boolean input = userinput.matches(".*\\d.*");
        if(input)
        {
            try
            {
                String inputASnumber = userinput.replaceAll("[^0-9]", "");
                int actualinput = Integer.parseInt(inputASnumber);
                return actualinput;
            }
            catch(NumberFormatException e)
            {
                System.out.println("Invalid Input.");
                return GetChoice(prompt);
            }
        }

        else
        {
            System.out.println("Invalid Input.");
            
            return GetChoice(prompt);
        }
    }






    // OVERLOADING IS HERE
    public static int GetChoice(String prompt, int min, int max)
    {

        System.out.print(prompt);
        String userinput;
        userinput = in.nextLine();
        int actualinput = 0;

        boolean input = userinput.matches(".*\\d.*");
        if(input)
        {
            try
            {
                String inputASnumber = userinput.replaceAll("[^0-9]", "");
                actualinput = Integer.parseInt(inputASnumber);
            }
            catch(NumberFormatException e)
            {
                System.out.println("Invalid Input.");
            }

            if(min <= actualinput && actualinput <= max)
            {
                return actualinput;
            }
            else
            {
                return GetChoice(prompt, min, max);
            }
        }

        else
        {
            System.out.println("Invalid Input.");
            
            return GetChoice(prompt, min, max);
        }
    }




    public static String GetInput(String prompt)
    {

    }
